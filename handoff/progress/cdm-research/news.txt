    (function ($) {
        $(function () {
            const urlSearchParams = new URLSearchParams(window.location.search);
            const params = Object.fromEntries(urlSearchParams.entries());
            var userToken = params.usertoken ? params.usertoken : params.userToken;
            if (!userToken) {
                userToken = "";
            }

            var xhr = new XMLHttpRequest();
            var sendParams = 'userToken=' + userToken;
            xhr.open('GET', '/sso/status' + "?" + sendParams, false);

            xhr.onload = function () {
                if (userToken) {
                    window.location.replace(params.ReturnURL ? params.ReturnURL : params.ReturnUrl);
                }
                if (xhr.status === 200) {
                    var response = JSON.parse(xhr.responseText);
                    if (response.IsSignedIn) {
                        document.getElementById("logged-in").style.display = "block";
                        document.getElementById("welcome").innerHTML = "Welcome, " + response.FirstName
                        document.getElementById("updateRedirectUrl").setAttribute("href", response.UpdateRedirectURL);
                        document.getElementById("updatePasswordRedirectUrl").setAttribute("href", response.UpdatePasswordRedirectURL);
                    }
                    else{
                        document.getElementById("logged-out").style.display = "block";
                    }
                }
            };
            xhr.setRequestHeader('Cache-Control', 'no-cache, no-store, must-revalidate');
            xhr.setRequestHeader('Pragma', 'no-cache');
            xhr.setRequestHeader('Expires', '0');
            xhr.send(null);
        });
    }(jQuery));

    $('#sign-out').on('click', () => {
        $.ajax({

            success: function (response) {
                window.location.replace(response);
            },
        });
    })
